\section*{Cardinality}
|S| of a finite set is the numbers of elements it contains. \\
For finite sets A and B it holds that $|A \cup B| = |A| + |B| - |A \cap B|$. \\
If A and B are disjoint, then $|A \cup B| = |A| + |B|$ \\
Let S be a finite set. Then $|P(S)| = 2^{|S|}$ \\
\textbf{Bijective Functions} 
\begin{itemize}[noitemsep,topsep=0pt,leftmargin=*]
    \item f is a function from A to B 
    \item \textit{injective}: if $x \neq y, \text{then }f(x) \neq f(y)$ %every element of A is mapped to an element of B. Different elements of A are mapped to different elements of B. 
    \item \textit{surjective}: $\forall y \in B$ there is an $x \in A$ with $f(x) = y$ 
\end{itemize}
%If there is a bijection from A to B there is one from B to A.
\textbf{Cardinality not larger}: Set A has cardinality less than or equal to the cardinality of set B ($|A| \leq |B|$, if there is an injective function from A to B. \\
\textbf{Equinumerous Sets}: Two sets A and B have the same cardinality ($|A| = |B|$) if there exists a bijection from A to B. \\
\textbf{Strictly smaller cardinality}: $|A| < |B|$ if there is an injective function $|A| \leq |B|$ and  $|A| \neq |B|$ \\
\textbf{Countable Sets} if $|A| \leq |\mathbb{N}_0|$ 
\begin{itemize}[noitemsep,topsep=0pt,leftmargin=*]
    \item Let A be a countable set. Every set B with $B \subseteq A$ is countable.
\end{itemize}
\textbf{Countably Infinite Sets} if $|A| = |\mathbb{N}_0|$ (there exists a bijection). Countable and Infinite 
\begin{itemize}[noitemsep,topsep=0pt,leftmargin=*]
    \item examples: even natural numbers, $n^2$ perfect squares, $\mathbb{Q}_+$, $\mathbb{Q}$. $\mathbb{Z}$ 
\end{itemize}
\textbf{Arbitrary unions}: Let M be a set of sets. The union $\bigcup_{S \in M}$ is the set with $x \in \bigcup_{S \in M} S$ with $x \in S$.
\begin{itemize}[noitemsep,topsep=0pt,leftmargin=*]
    \item Let M be a countable set of countable sets. Then $\bigcup_{S \in M}$ is countable.
    \item The set $B = \{ b \mid \text{ b is a binary tree }\}$ is countable
\end{itemize}
\textbf{Cantor's Theorem}: For every set S it holds that $|S| < |P(S)|$. Infinite Sets can have different cardinalities
